cmake_minimum_required(VERSION 3.16)
project(PetWings CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(SOURCES
  src/engine.cpp
  src/game.cpp
  src/boss.cpp
  src/title.cpp
  src/mainapp.cpp
  src/platform_sdl.cpp
  third_party/stb_vorbis.c
)
set_source_files_properties(third_party/stb_vorbis.c PROPERTIES COMPILE_OPTIONS "-w")

add_executable(petwings ${SOURCES})

if(EMSCRIPTEN)
  set(DATA_DIR "${CMAKE_SOURCE_DIR}/data" CACHE PATH "Prepared game data")
  # absolute address used by the link-preview tags (og:image must be an absolute URL)
  set(PETWINGS_PUBLIC_URL "https://petwings.mojavelabs.cloud" CACHE STRING "Public URL of the web build")
  configure_file("${CMAKE_SOURCE_DIR}/web/shell.html" "${CMAKE_BINARY_DIR}/shell.html" @ONLY)
  set(USE_FLAGS "-sUSE_SDL=2")
  target_compile_options(petwings PRIVATE ${USE_FLAGS})
  target_link_options(petwings PRIVATE ${USE_FLAGS}
    -sALLOW_MEMORY_GROWTH=1
    -sINVOKE_RUN=0
    -sEXPORTED_RUNTIME_METHODS=callMain,stringToNewUTF8,UTF8ToString
    -sEXPORTED_FUNCTIONS=_main,_malloc,_free
    -sENVIRONMENT=web
    -sMINIFY_HTML=0  # keep the link-preview meta tags readable for every scraper
    "--preload-file=${DATA_DIR}@/data"
    "--exclude-file=${DATA_DIR}/web/*"
    "--shell-file=${CMAKE_BINARY_DIR}/shell.html")
  set_target_properties(petwings PROPERTIES OUTPUT_NAME "index" SUFFIX ".html")
  set_property(TARGET petwings APPEND PROPERTY LINK_DEPENDS "${CMAKE_BINARY_DIR}/shell.html")
  # page assets next to index.html: PWA manifest and icons (generated by tools/prepare_data.py)
  add_custom_command(TARGET petwings POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/web/manifest.webmanifest" "$<TARGET_FILE_DIR:petwings>/"
    COMMAND ${CMAKE_COMMAND} -E copy "${DATA_DIR}/web/icon-32.png" "${DATA_DIR}/web/icon-180.png"
            "${DATA_DIR}/web/icon-192.png" "${DATA_DIR}/web/icon-512.png" "${DATA_DIR}/web/og-image.png"
            "$<TARGET_FILE_DIR:petwings>/")
else()
  find_package(SDL2 REQUIRED)
  target_link_libraries(petwings PRIVATE SDL2::SDL2 m)
  target_compile_options(petwings PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
